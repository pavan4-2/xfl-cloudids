Step 3: Run the Flower Federated Simulation Live
If they specifically ask to see the Federated Learning (Flower) simulation run, execute this script:  
PDF

PowerShell

python -c "
import flwr as fl
from flwr.client import ClientApp, NumPyClient
from flwr.server import ServerApp, ServerConfig, ServerAppComponents
from flwr.server.strategy import FedAvg
from flwr.simulation import run_simulation

class SmokeClient(NumPyClient):
    def get_parameters(self, config): return []
    def fit(self, parameters, config): return [], 10, {}
    def evaluate(self, parameters, config): return 0.0, 10, {'accuracy': 1.0}

client = ClientApp(client_fn=lambda ctx: SmokeClient().to_client())
server = ServerApp(server_fn=lambda ctx: ServerAppComponents(strategy=FedAvg(), config=ServerConfig(num_rounds=1)))

print('--- EXECUTING F0 FLOWER 3-CLIENT SIMULATION ---')
run_simulation(server_app=server, client_app=client, num_supernodes=3)
print('--- FLOWER SIMULATION SUCCESSFUL ---')
"
